% --- SLIDE 3: 5.1 EQUILÍBRIO ---
\begin{frame}{5.1 Uso da Primeira Lei de Newton: Partículas em Equilíbrio}
    \begin{columns}[T]
        \column{0.50\textwidth}
        \begin{block}{Condição de Equilíbrio ($\vec{a}=0$)}
            \small
            $$\sum \vec{F} = 0 \iff \begin{cases} \sum F_x = 0 \\ \sum F_y = 0 \end{cases}$$
        \end{block}
        \begin{alertblock}{Regra Fundamental do DCL}
            \small
            Desenhe \textbf{apenas} forças externas exercidas \textbf{SOBRE} o corpo isolado!
        \end{alertblock}

        \column{0.46\textwidth}
        \begin{block}{Passos Estratégicos}
            \small
            \begin{enumerate}\setlength{\itemsep}{2pt}
                \item Isolamento do corpo e eixos ($x,y$).
                \item Diagrama do Corpo Livre (DCL).
                \item Decomposição vectorial das forças.
                \item Resolução do sistema linear.
            \end{enumerate}
        \end{block}
    \end{columns}
\end{frame}

% --- SLIDE 4: EXEMPLO 5.1 ---
\begin{frame}{Exemplo 5.1: Equilíbrio 1D --- Ginasta em Corda sem Massa}
    \begin{block}{Enunciado do Problema}
        \small
        Uma ginasta de massa $m_G = 50{,}0\text{ kg}$ está subindo em uma corda leve sem massa presa ao teto. Qual o seu peso? Qual força a corda exerce nela? Qual a tensão no teto?
    \end{block}
    \begin{columns}[T]
        \column{0.40\textwidth}
        \begin{center}
        \begin{tikzpicture}[scale=0.62]
            \fill[pattern=north east lines] (0,3) rectangle (2,3.2);
            \draw[thick] (0,3) -- (2,3);
            \draw[ultra thick,brown] (1,3) -- (1,1.5);
            \fill[navyblue] (1,1.5) circle (0.15);
            \draw[ultra thick,-{Latex},emeraldgreen] (1,1.5) -- (1,2.8) node[right] {\small $\vec{T}_{\text{corda}}$};
            \draw[ultra thick,-{Latex},redaccent] (1,1.5) -- (1,0.2) node[right] {\small $\vec{p}_G$};
            \node at (1,-0.3) {\small DCL da Ginasta};
        \end{tikzpicture}
        \end{center}
        \column{0.54\textwidth}
        \begin{exampleblock}{Resolução}
            \small
            \begin{itemize}\setlength{\itemsep}{3pt}
                \item \textbf{Peso}: $p_G = m_G g = (50{,}0)(9{,}80) = 490\text{ N}$
                \item \textbf{Força da Corda}: $T_{\text{corda}} - p_G = 0 \implies T_{\text{corda}} = 490\text{ N}$
                \item \textbf{Tensão no Teto}: $T_{\text{teto}} - T_{\text{corda}} = 0 \implies T_{\text{teto}} = 490\text{ N}$
            \end{itemize}
        \end{exampleblock}
    \end{columns}
\end{frame}

    \begin{block}{Enunciado do Problema}
        \small

    \end{block}
    \begin{columns}[T]
        \column{0.40\textwidth}
        \begin{center}

            \draw[ultra thick,-{Latex},redaccent] (1,1.5) -- (1,0.3) node[right] {\small $\vec{p}_G$};
            \node at (1,-0.2) {\small DCLs Separados};

                \item \textbf{Tensão no Topo}: Suporta ginasta e corda $\implies T_{\text{topo}} = (m_G + m_C)g = (52{,}0)(9{,}80) = 509{,}6\text{ N}$.
            \end{itemize}
        \end{exampleblock}
    \end{columns}
\end{frame}

\begin{frame}{Exemplo 5.3: Equilíbrio 2D --- Motor Suspenso por Correntes}
    \begin{block}{Enunciado do Problema}
        \small
        Um motor de peso $p$ está suspenso por um anel $O$ ligado a três correntes ($60^\circ$ no teto). Ache as tensões $T_1$, $T_2$ e $T_3$.
    \end{block}
    \begin{columns}[T]
        \column{0.40\textwidth}
        \begin{center}
        \begin{tikzpicture}[scale=0.65]
            \fill[pattern=north east lines] (0,3) -- (3,3) -- (3,3.2) -- (0,3.2) -- cycle;
            \fill[pattern=north east lines] (0,0) -- (0,3) -- (-0.2,3) -- (-0.2,0) -- cycle;
            \draw[thick] (0,3) -- (3,3); \draw[thick] (0,0) -- (0,3);
            \fill[navyblue] (1.5,1.5) circle (0.1) node[below left] {$O$};
            \draw[ultra thick,-{Latex},redaccent] (1.5,1.5) -- (1.5,0.2) node[right] {\small $T_1 = p$};
            \draw[ultra thick,-{Latex},golddark] (1.5,1.5) -- (0.2,1.5) node[above] {\small $T_2$};
            \draw[ultra thick,-{Latex},emeraldgreen] (1.5,1.5) -- (2.8,2.8) node[right] {\small $T_3$};
            \draw[dashed,gray] (1.5,1.5) -- (2.8,1.5);
            \node at (2.2,1.8) {\small $60^\circ$};
        \end{tikzpicture}
        \end{center}
        \column{0.54\textwidth}
        \begin{exampleblock}{Resolução no Anel $O$}
            \small
            \begin{itemize}\setlength{\itemsep}{3pt}
                \item \textbf{Eixo $y$}: $T_3 \sin 60^\circ - p = 0 \implies T_3 = \frac{p}{\sin 60^\circ} \approx 1{,}155 \, p$
                \item \textbf{Eixo $x$}: $T_3 \cos 60^\circ - T_2 = 0 \implies T_2 = 0{,}577 \, p$
                \item \textbf{Corda Vertical}: $T_1 = p$
            \end{itemize}
        \end{exampleblock}
    \end{columns}
\end{frame}

\begin{frame}{Exemplo 5.4: Plano Inclinado sem Atrito em Equilíbrio (Carro na Rampa)}
    \begin{block}{Enunciado do Problema}
        \small
        Um carro de peso $p$ repousa em uma rampa inclinada de $\alpha$. Um cabo impede o deslize. Ache a tensão $T$ no cabo e a força normal $n$.
    \end{block}
    \begin{columns}[T]
        \column{0.40\textwidth}
        \begin{center}
        \begin{tikzpicture}[scale=0.68]
            \draw[thick] (0,0) -- (3.5,0) -- (3.5,2) -- cycle;
            \node at (0.8,0.2) {\small $\alpha$};
            \fill[cyandark] (1.75,1) circle (0.12);
            \draw[gray,dashed,-{Latex}] (1.75,1) -- (3.0,1.7) node[above] {$x$};
            \draw[gray,dashed,-{Latex}] (1.75,1) -- (1.05,2.2) node[left] {$y$};
            \draw[ultra thick,-{Latex},redaccent] (1.75,1) -- (2.61,1.5) node[above] {\small $\vec{T}$};
            \draw[ultra thick,-{Latex},emeraldgreen] (1.75,1) -- (1.0,2.3) node[above] {\small $\vec{n}$};
            \draw[ultra thick,-{Latex},golddark] (1.75,1) -- (1.75,-0.5) node[below] {\small $\vec{p}$};
        \end{tikzpicture}
        \end{center}
        \column{0.54\textwidth}
        \begin{exampleblock}{Resolução com Eixos Rotacionados}
            \small
            Rotacionando os eixos paralelamente ao plano inclinado:
            \begin{align*}
                \sum F_x &= T - p \sin\alpha = 0 \implies T = p \sin\alpha \\
                \sum F_y &= n - p \cos\alpha = 0 \implies n = p \cos\alpha
            \end{align*}
        \end{exampleblock}
    \end{columns}
\end{frame}

% --- SLIDE 9: 5.2 SEGUNDA LEI ---
\begin{frame}{5.2 Uso da Segunda Lei de Newton: Dinâmica das Partículas}
    \begin{columns}[T]
        \column{0.50\textwidth}
        \begin{block}{Segunda Lei de Newton ($\vec{a} \neq 0$)}
            \small
            $$\sum \vec{F} = m \vec{a} \iff \begin{cases} \sum F_x = m a_x \\ \sum F_y = m a_y \end{cases}$$
        \end{block}
        \begin{alertblock}{ALERTA CRUCIAL!}
            \small
            $m\vec{a}$ \textbf{NÃO É UMA FORÇA}! Nunca desenhe $m\vec{a}$ dentro do DCL. Desenhe $\vec{a}$ ao lado!
        \end{alertblock}

        \column{0.46\textwidth}
        \begin{center}
        \begin{tikzpicture}[scale=0.75]
            \node[draw=redaccent,thick,fill=redaccent!10,inner sep=4pt] at (0,1.0) {
                \shortstack{\textbf{ERRADO!}\\[2pt] Incluir $m\vec{a}$ no DCL}
            };
            \node[draw=emeraldgreen,thick,fill=emeraldgreen!10,inner sep=4pt] at (0,-0.8) {
                \shortstack{\textbf{CERTO!}\\[2pt] Desenhar $\vec{a}$ ao lado}
            };
            \draw[thick,-{Latex},blue] (1.4,-0.8) -- (2.3,-0.8) node[right] {$\vec{a}$};
        \end{tikzpicture}
        \end{center}
    \end{columns}
\end{frame}

\begin{frame}{Exemplo 5.6: Aceleração em Linha Reta --- Carro de Corridas}
    \begin{block}{Enunciado do Problema}
        \small
        Um carro de corrida de $m = 1000\text{ kg}$ acelera em linha reta de $0$ a $30\text{ m/s}$ em $6{,}0\text{ s}$. Qual a força resultante horizontal exercida sobre ele?
    \end{block}
    \begin{columns}[T]
        \column{0.40\textwidth}
        \begin{center}
        \begin{tikzpicture}[scale=0.65]
            \fill[navyblue] (1.5,1) circle (0.15);
            \draw[ultra thick,-{Latex},emeraldgreen] (1.5,1) -- (3.0,1) node[above] {\small $\vec{F}_x$};
            \draw[ultra thick,-{Latex},cyandark] (1.5,1) -- (1.5,2.2) node[right] {\small $\vec{n}$};
            \draw[ultra thick,-{Latex},golddark] (1.5,1) -- (1.5,-0.2) node[right] {\small $\vec{p}$};
            \draw[thick,-{Latex},blue] (1.5,2.6) -- (2.8,2.6) node[above] {\small $\vec{a}_x$};

        \end{center}
        \column{0.54\textwidth}
        \begin{exampleblock}{Resolução}
            \small
            \begin{enumerate}\setlength{\itemsep}{3pt}
                \item Aceleração média: $a_x = \frac{\Delta v_x}{\Delta t} = \frac{30 - 0}{6{,}0} = 5{,}0\text{ m/s}^2$
                \item Segunda Lei: $F_x = m a_x = (1000\text{ kg})(5{,}0\text{ m/s}^2) = 5000\text{ N}$
            \end{enumerate}
        \end{exampleblock}
    \end{columns}
\end{frame}

\begin{frame}{Exemplo 5.7: Força Variável com o Tempo em Foguete de Teste}
    \begin{block}{Enunciado do Problema}
        \small
        Um foguete de massa $m$ parte do repouso em $t=0$ sob ação de força horizontal $F_x(t) = A t^2$. Ache a velocidade $v_x(t)$ e a posição $x(t)$.
    \end{block}
    \begin{columns}[T]
        \column{0.40\textwidth}
        \begin{center}
        \begin{tikzpicture}[scale=0.65]
            \draw[-{Latex}] (0,0) -- (3,0) node[below] {\small $t$};
            \draw[-{Latex}] (0,0) -- (0,2.5) node[left] {\small $F_x$};
            \draw[ultra thick,cyandark,domain=0:2.2,samples=30] plot (\x, {0.4*\x*\x});
            \node at (1.8,2.0) {\small $A t^2$};
        \end{tikzpicture}
        \end{center}
        \column{0.54\textwidth}
        \begin{exampleblock}{Integração}
            \small
            $$a_x(t) = \frac{F_x(t)}{m} = \frac{A}{m} t^2$$
            $$v_x(t) = \int_0^t a_x(t') dt' = \frac{A}{3m} t^3$$
            $$x(t) = \int_0^t v_x(t') dt' = \frac{A}{12m} t^4$$
        \end{exampleblock}
    \end{columns}
\end{frame}

\begin{frame}{Exemplo 5.8: Elevador em Frenagem --- Tensão no Cabo}
    \begin{block}{Enunciado do Problema}
        \small
        Um elevador de $m = 800\text{ kg}$ descendo a $10{,}0\text{ m/s}$ para em $25{,}0\text{ m}$. Ache a tensão $T$ no cabo durante a frenagem.
    \end{block}
    \begin{columns}[T]
        \column{0.40\textwidth}
        \begin{center}
        \begin{tikzpicture}[scale=0.68]
            \fill[navyblue] (1.5,1.5) circle (0.15);
            \draw[ultra thick,-{Latex},emeraldgreen] (1.5,1.5) -- (1.5,2.9) node[left] {\small $\vec{T}$};
            \draw[ultra thick,-{Latex},redaccent] (1.5,1.5) -- (1.5,0.2) node[left] {\small $\vec{p}$};
            \draw[ultra thick,-{Latex},blue] (2.3,1.5) -- (2.3,2.5) node[right] {\small $\vec{a}_y$};
            \draw[thick,-{Latex},gray] (0.7,1.5) -- (0.7,0.5) node[left] {\small $\vec{v}_y$};
        \end{tikzpicture}
        \end{center}
        \column{0.54\textwidth}
        \begin{exampleblock}{Resolução}
            \small
            \begin{itemize}\setlength{\itemsep}{3pt}
                \item Aceleração: $v_y^2 = v_{0y}^2 + 2a_y \Delta y \implies a_y = +2{,}0\text{ m/s}^2$ (para cima)
                \item Segunda Lei: $\sum F_y = T - mg = m a_y$
                \item $T = m(g + a_y) = 800(9{,}80 + 2{,}0) = 9440\text{ N}$
            \end{itemize}
        \end{exampleblock}
    \end{columns}
\end{frame}

    \begin{columns}[T]

\begin{frame}{Exemplo 5.10: Tobogã Descendo Encosta sem Atrito}
    \begin{block}{Enunciado do Problema}
        \small
        Um tobogã cheio de estudantes (peso $p$) escorrega para baixo em encosta sem atrito com inclinação constante $\alpha$. Qual é a sua aceleração?
    \end{block}
    \begin{columns}[T]
        \column{0.40\textwidth}
        \begin{center}
        \begin{tikzpicture}[scale=0.68]
            \draw[thick] (0,0) -- (3.5,0) -- (3.5,2) -- cycle;
            \fill[cyandark] (1.75,1) circle (0.12);
            \draw[ultra thick,-{Latex},emeraldgreen] (1.75,1) -- (1.0,2.3) node[above] {\small $\vec{n}$};
            \draw[ultra thick,-{Latex},golddark] (1.75,1) -- (1.75,-0.5) node[below] {\small $\vec{p}$};
            % Aceleração aponta RAMPA ABAIXO (para baixo e para a esquerda, rumo à base)
            \draw[ultra thick,-{Latex},blue] (1.3,0.74) -- (0.44,0.24) node[below left] {\small $\vec{a}_x$};
        \end{tikzpicture}
        \end{center}
        \column{0.54\textwidth}
        \begin{exampleblock}{Resolução}
            \small
            $$\sum F_x = mg \sin\alpha = m a_x \implies a_x = g \sin\alpha \quad (\text{rampa abaixo})$$
            $$\sum F_y = n - mg \cos\alpha = 0 \implies n = mg \cos\alpha$$
            \textbf{Conclusão}: A aceleração independe da massa total do tobogã!
        \end{exampleblock}

\begin{frame}{5.3 Forças de Atrito: Estático, Cinético e Rolamento}

            \small
            \begin{itemize}\setlength{\itemsep}{3pt}
                \item \textbf{Estático ($f_s$)}: $f_s \le \mu_s n \implies (f_s)_{\text{máx}} = \mu_s n$

            \end{itemize}
        \end{block}

    \end{columns}
\end{frame}

\begin{frame}{Exemplo 5.24: Movimento Circular em Círculo Vertical (Roda-Gigante)}
    \begin{block}{Enunciado do Problema}
        \small
        Um passageiro na roda-gigante de raio $R$ move-se em círculo vertical com velocidade $v$. Ache a normal $n$ no topo e na base.
    \end{block}
    \begin{columns}[T]
        \column{0.40\textwidth}
        \begin{center}
        \begin{tikzpicture}[scale=0.65]
            \draw[dashed] (1.5,1.5) circle (1.2);
            % Topo
            \fill[navyblue] (1.5,2.7) circle (0.1);
            \draw[thick,-{Latex},emeraldgreen] (1.5,2.7) -- (1.5,3.3) node[above] {\tiny $n_T$};
            \draw[thick,-{Latex},redaccent] (1.5,2.7) -- (1.5,2.0) node[left] {\tiny $p$};
            \draw[thick,-{Latex},blue] (1.8,2.7) -- (1.8,2.1) node[right] {\tiny $\vec{a}_{\text{rad}}$};
            % Base
            \fill[navyblue] (1.5,0.3) circle (0.1);
            \draw[thick,-{Latex},emeraldgreen] (1.5,0.3) -- (1.5,1.1) node[above] {\tiny $n_B$};
            \draw[thick,-{Latex},redaccent] (1.5,0.3) -- (1.5,-0.3) node[below] {\tiny $p$};
            \draw[thick,-{Latex},blue] (1.8,0.3) -- (1.8,0.9) node[right] {\tiny $\vec{a}_{\text{rad}}$};
        \end{tikzpicture}
        \end{center}
        \column{0.54\textwidth}
        \begin{exampleblock}{Resolução}
            \small
            \begin{itemize}\setlength{\itemsep}{3pt}
                \item \textbf{Topo}: $n_T - mg = -m \frac{v^2}{R} \implies n_T = m\left(g - \frac{v^2}{R}\right)$
                \item \textbf{Base}: $n_B - mg = +m \frac{v^2}{R} \implies n_B = m\left(g + \frac{v^2}{R}\right)$
            \end{itemize}
        \end{exampleblock}
    \end{columns}
\end{frame}
