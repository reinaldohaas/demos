% --- SLIDE 3: 5.1 CONCEITO DE EQUILÍBRIO E DCL ---
\begin{frame}{5.1 Uso da Primeira Lei de Newton: Partículas em Equilíbrio}
    \begin{columns}
        \column{0.55\textwidth}
        \begin{block}{Condição de Equilíbrio ($\vec{a}=0$)}
            Um corpo está em equilíbrio se permanece em repouso ou em movimento retilíneo uniforme (MRU):
            $$\sum \vec{F} = 0 \iff \begin{cases} \sum F_x = 0 \\ \sum F_y = 0 \end{cases}$$
        \end{block}

        \begin{alertblock}{Regra Fundamental para o DCL}
            Desenhe \textbf{apenas} as forças externas exercidas \textbf{SOBRE} o corpo isolado. Nunca inclua forças que o corpo exerce sobre outros objetos!
        \end{alertblock}

        \column{0.45\textwidth}
        \begin{block}{Passos da Estratégia 5.1}
            \begin{enumerate}
                \item Isolamento e eixos ($x,y$).
                \item Diagrama do Corpo Livre.
                \item Decomposição das forças.
                \item Resolução de $\sum F_x = 0$ e $\sum F_y = 0$.
            \end{enumerate}
        \end{block}
    \end{columns}
\end{frame}

% --- SLIDE 4: EXEMPLO 5.1 - GINASTA NA CORDA ---
\begin{frame}{Exemplo 5.1: Equilíbrio 1D --- Tensão em Corda sem Massa}
    \begin{block}{Enunciado do Problema}
        Uma ginasta com massa $m_G = 50{,}0\text{ kg}$ está subindo em uma corda presa ao teto de um ginásio. A massa da corda em si é desprezível. Qual é o peso da ginasta? Qual força a corda exerce sobre ela? Qual a tensão na extremidade superior da corda?
    \end{block}

    \begin{columns}
        \column{0.45\textwidth}
        \begin{center}
        \begin{tikzpicture}[scale=0.85]
            \fill[pattern=north east lines] (0,3) rectangle (2,3.2);
            \draw[thick] (0,3) -- (2,3);
            \draw[ultra thick,brown] (1,3) -- (1,1.5);
            \fill[navyblue] (1,1.5) circle (0.15);
            % DCL
            \draw[ultra thick,-{Latex},emeraldgreen] (1,1.5) -- (1,2.8) node[right] {\small $\vec{T}_{\text{corda}}$};
            \draw[ultra thick,-{Latex},redaccent] (1,1.5) -- (1,0.2) node[right] {\small $\vec{p}_G = m_G \vec{g}$};
            \node at (1,-0.3) {\small DCL da Ginasta};
        \end{tikzpicture}
        \end{center}

        \column{0.55\textwidth}
        \begin{exampleblock}{Resolução}
            \begin{enumerate}
                \item **Peso da Ginasta**:
                $$p_G = m_G g = (50{,}0\text{ kg})(9{,}80\text{ m/s}^2) = 490\text{ N}$$
                \item **Força exercida pela Corda**:
                $$\sum F_y = T_{\text{corda}} - p_G = 0 \implies T_{\text{corda}} = 490\text{ N}$$
                \item **Tensão no Teto (Corda sem massa)**:
                $$\sum F_y = T_{\text{teto}} - T_{\text{corda}} = 0 \implies T_{\text{teto}} = 490\text{ N}$$
            \end{enumerate}
        \end{exampleblock}
    \end{columns}
\end{frame}

% --- SLIDE 5: EXEMPLO 5.3 - MOTOR SUSPENSO EM 2D ---
\begin{frame}{Exemplo 5.3: Equilíbrio 2D --- Motor Suspenso por Correntes}
    \begin{block}{Enunciado do Problema}
        Um motor de automóvel com peso $p$ está suspenso por uma corrente ligada por um anel $O$ a duas outras correntes, uma amarrada ao teto ($60^\circ$ com a horizontal) e a outra presa à parede. Ache as tensões $T_1$, $T_2$ e $T_3$ nas três correntes em função de $p$.
    \end{block}

    \begin{columns}
        \column{0.45\textwidth}
        \begin{center}
        \begin{tikzpicture}[scale=0.8]
            % Anel O
            \fill[navyblue] (2,1.5) circle (0.1) node[below left] {$O$};
            % DCL no Anel
            \draw[ultra thick,-{Latex},redaccent] (2,1.5) -- (2,0.2) node[right] {\small $T_1 = p$};
            \draw[ultra thick,-{Latex},golddark] (2,1.5) -- (0.5,1.5) node[above] {\small $T_2$};
            \draw[ultra thick,-{Latex},emeraldgreen] (2,1.5) -- (3.3,2.8) node[right] {\small $T_3$};
            \draw[dashed,gray] (2,1.5) -- (3.3,1.5);
            \node at (2.7,1.8) {\small $60^\circ$};
            \node at (2,-0.3) {\small DCL no Anel $O$};
        \end{tikzpicture}
        \end{center}

        \column{0.55\textwidth}
        \begin{exampleblock}{Resolução no Anel $O$}
            \begin{itemize}
                \item **Motor**: $\sum F_y = T_1 - p = 0 \implies T_1 = p$
                \item **Anel $O$ (Eixo $y$)**:
                $$\sum F_y = T_3 \sin 60^\circ - T_1 = 0 \implies T_3 = \frac{p}{\sin 60^\circ} \approx 1{,}155 \, p$$
                \item **Anel $O$ (Eixo $x$)**:
                $$\sum F_x = T_3 \cos 60^\circ - T_2 = 0 \implies T_2 = 0{,}577 \, p$$
            \end{itemize}
        \end{exampleblock}
    \end{columns}
\end{frame}

% --- SLIDE 6: EXEMPLO 5.4 - PLANO INCLINADO EM EQUILÍBRIO ---
\begin{frame}{Exemplo 5.4: Plano Inclinado sem Atrito em Equilíbrio}
    \begin{block}{Enunciado do Problema}
        Um carro de peso $p$ está em repouso sobre a rampa de um rebocador inclinada de um ângulo $\alpha$. Somente um cabo impede o carro de deslizar rampa abaixo. Ache a tensão $T$ no cabo e a força normal $n$ exercida pela rampa.
    \end{block}

    \begin{columns}
        \column{0.45\textwidth}
        \begin{center}
        \begin{tikzpicture}[scale=0.85]
            \draw[thick] (0,0) -- (3.5,0) -- (3.5,2) -- cycle;
            \node at (0.8,0.2) {\small $\alpha$};
            % Eixos inclinados
            \draw[gray,dashed,-{Latex}] (1.75,1) -- (3.25,1.86) node[above] {$x$};
            \draw[gray,dashed,-{Latex}] (1.75,1) -- (0.89,2.5) node[left] {$y$};
            \fill[cyandark] (1.75,1) circle (0.12);
            % Forças
            \draw[ultra thick,-{Latex},redaccent] (1.75,1) -- (2.61,1.5) node[above] {\small $\vec{T}$};
            \draw[ultra thick,-{Latex},emeraldgreen] (1.75,1) -- (1.0,2.3) node[above] {\small $\vec{n}$};
            \draw[ultra thick,-{Latex},golddark] (1.75,1) -- (1.75,-0.5) node[below] {\small $\vec{p}$};
            \node at (1.75,-0.9) {\small DCL do Carro};
        \end{tikzpicture}
        \end{center}

        \column{0.55\textwidth}
        \begin{exampleblock}{Resolução com Eixos Rotacionados}
            Decomposição do peso $\vec{p}$:
            $$p_x = -p \sin\alpha, \quad p_y = -p \cos\alpha$$
            Equações de Equilíbrio:
            \begin{align*}
                \sum F_x &= T - p \sin\alpha = 0 \implies T = p \sin\alpha \\
                \sum F_y &= n - p \cos\alpha = 0 \implies n = p \cos\alpha
            \end{align*}
        \end{exampleblock}
    \end{columns}
\end{frame}

% --- SLIDE 7: 5.2 SEGUNDA LEI (DINÂMICA) E ALERTA ---
\begin{frame}{5.2 Uso da Segunda Lei de Newton: Dinâmica das Partículas}
    \begin{columns}
        \column{0.55\textwidth}
        \begin{block}{Segunda Lei de Newton ($\vec{a} \neq 0$)}
            Quando a força resultante sobre a partícula não é nula, o corpo acelera com $\vec{a}$:
            $$\sum \vec{F} = m \vec{a} \iff \begin{cases} \sum F_x = m a_x \\ \sum F_y = m a_y \end{cases}$$
        \end{block}

        \begin{alertblock}{ALERTA DIDÁTICO CRUCIAL!}
            \begin{itemize}
                \item O produto $m\vec{a}$ \textbf{NÃO É UMA FORÇA}!
                \item Nunca desenhe o vetor $m\vec{a}$ dentro do Diagrama do Corpo Livre (DCL).
                \item O vetor aceleração $\vec{a}$ deve ser desenhado \textbf{AO LADO} do corpo.
            \end{itemize}
        \end{alertblock}

        \column{0.45\textwidth}
        \begin{center}
        \begin{tikzpicture}[scale=0.9]
            \node[draw=redaccent,thick,fill=redaccent!10,inner sep=4pt] at (0,1.2) {
                \shortstack{\textbf{ERRADO!}\\[2pt] Incluir $m\vec{a}$ no DCL}
            };
            \node[draw=emeraldgreen,thick,fill=emeraldgreen!10,inner sep=4pt] at (0,-1.0) {
                \shortstack{\textbf{CERTO!}\\[2pt] Desenhar $\vec{a}$ ao lado}
            };
            \draw[thick,-{Latex},blue] (1.5,-1.0) -- (2.5,-1.0) node[right] {$\vec{a}$};
        \end{tikzpicture}
        \end{center}
    \end{columns}
\end{frame}

% --- SLIDE 8: EXEMPLO 5.8 & 5.9 - ELEVADOR E PESO APARENTE ---
\begin{frame}{Exemplo 5.8 \& 5.9: Elevador e Peso Aparente}
    \begin{block}{Enunciado do Problema}
        Um elevador de $m = 800\text{ kg}$ descendo a $10{,}0\text{ m/s}$ para em $25{,}0\text{ m}$. Ache a tensão no cabo. Qual é a leitura da balança para uma passageira de $50{,}0\text{ kg}$ durante a frenagem ($a_y = +2{,}0\text{ m/s}^2$)?
    \end{block}

    \begin{columns}
        \column{0.45\textwidth}
        \begin{center}
        \begin{tikzpicture}[scale=0.8]
            \fill[navyblue] (1.5,1.5) circle (0.15);
            \draw[ultra thick,-{Latex},emeraldgreen] (1.5,1.5) -- (1.5,3.0) node[right] {\small $\vec{n}$ (ou $\vec{T}$)};
            \draw[ultra thick,-{Latex},redaccent] (1.5,1.5) -- (1.5,0.0) node[right] {\small $\vec{p} = m\vec{g}$};
            \draw[ultra thick,-{Latex},blue] (2.8,1.5) -- (2.8,2.5) node[above] {$\vec{a}_y$};
            \node at (1.5,-0.4) {\small DCL na Balança};
        \end{tikzpicture}
        \end{center}

        \column{0.55\textwidth}
        \begin{exampleblock}{Resolução}
            \begin{enumerate}
                \item **Aceleração de Frenagem**:
                $$v_y^2 = v_{0y}^2 + 2a_y \Delta y \implies 0 = (-10)^2 + 2a_y(-25) \implies a_y = +2{,}0\text{ m/s}^2$$
                \item **Tensão no Cabo do Elevador**:
                $$T = m(g + a_y) = 800(9{,}80 + 2{,}0) = 9440\text{ N}$$
                \item **Peso Aparente da Passageira ($n$)**:
                $$n = m_G(g + a_y) = 50(9{,}80 + 2{,}0) = 590\text{ N}$$
            \end{enumerate}
        \end{exampleblock}
    \end{columns}
\end{frame}

\begin{frame}{Exemplo 5.10: Tobogã Descendo Encosta sem Atrito}
    \begin{block}{Enunciado do Problema}
        Um tobogã cheio de estudantes (peso total $p$) escorrega para baixo em uma encosta coberta de neve sem atrito com inclinação constante $\alpha$. Qual é a aceleração do tobogã?
    \end{block}

    \begin{columns}
        \column{0.45\textwidth}
        \begin{center}
        \begin{tikzpicture}[scale=0.85]
            \draw[thick] (0,0) -- (3.5,0) -- (3.5,2) -- cycle;
            \fill[cyandark] (1.75,1) circle (0.12);
            \draw[ultra thick,-{Latex},emeraldgreen] (1.75,1) -- (1.0,2.3) node[above] {\small $\vec{n}$};
            \draw[ultra thick,-{Latex},golddark] (1.75,1) -- (1.75,-0.5) node[below] {\small $\vec{p}$};
            \draw[ultra thick,-{Latex},blue] (2.2,1.26) -- (3.06,1.76) node[above] {\small $\vec{a}_x$};
            \node at (1.75,-0.9) {\small DCL do Tobogã};
        \end{tikzpicture}
        \end{center}

        \column{0.55\textwidth}
        \begin{exampleblock}{Resolução}
            Equações de movimento no plano inclinado:
            $$\sum F_x = mg \sin\alpha = m a_x \implies a_x = g \sin\alpha$$
            $$\sum F_y = n - mg \cos\alpha = 0 \implies n = mg \cos\alpha$$
            \textbf{Conclusão}: A aceleração independe da massa total do tobogã ou do número de passageiros!
        \end{exampleblock}
    \end{columns}
\end{frame}

% --- SLIDE 10: EXEMPLO 5.12 - MÁQUINA DE ATWOOD HORIZONTAL ---
\begin{frame}{Exemplo 5.12: Dois Corpos Acoplados (Máquina de Atwood Horizontal)}
    \begin{block}{Enunciado do Problema}
        Um cavaleiro de massa $m_1$ desliza sobre um trilho de ar horizontal sem atrito ligado a um peso de massa $m_2$ suspenso por um fio leve sem massa que passa por uma polia sem atrito. Calcule a aceleração do sistema e a tensão no fio.
    \end{block}

    \begin{columns}
        \column{0.45\textwidth}
        \begin{center}
        \begin{tikzpicture}[scale=0.75]
            \draw[thick] (0,2) -- (2.5,2) -- (2.5,0);
            \draw[thick] (2.5,2) circle (0.2);
            \draw[thick,fill=cyan!30] (0.5,2) rectangle (1.5,2.6) node[pos=0.5] {$m_1$};
            \draw[thick] (1.5,2.3) -- (2.5,2.2);
            \draw[thick] (2.7,2.0) -- (2.7,0.8);
            \draw[thick,fill=orange!30] (2.4,0.2) rectangle (3.0,0.8) node[pos=0.5] {$m_2$};
            \draw[thick,-{Stealth},blue] (1.0,2.9) -- (1.8,2.9) node[above] {$\vec{a}$};
            \draw[thick,-{Stealth},blue] (3.3,0.8) -- (3.3,0.2) node[right] {$\vec{a}$};
        \end{tikzpicture}
        \end{center}

        \column{0.55\textwidth}
        \begin{exampleblock}{Resolução}
            Equações de Segunda Lei de Newton:
            $$\text{Corpo 1 (trilho): } T = m_1 a$$
            $$\text{Corpo 2 (suspenso): } m_2 g - T = m_2 a$$
            Somando as equações:
            $$a = \left(\frac{m_2}{m_1 + m_2}\right) g \quad \text{e} \quad T = \left(\frac{m_1 m_2}{m_1 + m_2}\right) g$$
        \end{exampleblock}
    \end{columns}
\end{frame}

% --- SLIDE 11: 5.3 FORÇAS DE ATRITO ---
\begin{frame}{5.3 Forças de Atrito: Estático vs Cinético}
    \begin{columns}
        \column{0.55\textwidth}
        \begin{block}{Modelagem das Forças de Atrito}
            \begin{itemize}
                \item \textbf{Atrito Estático ($f_s$)}: Previne o deslizamento inicial:
                $$f_s \le \mu_s n \implies (f_s)_{\text{máx}} = \mu_s n$$
                \item \textbf{Atrito Cinético ($f_c$)}: Atua durante o deslizamento:
                $$f_c = \mu_c n \quad (\text{com } \mu_c < \mu_s)$$
            \end{itemize}
        \end{block}

        \column{0.45\textwidth}
        \begin{center}
        \begin{tikzpicture}[scale=0.75]
            \draw[-{Latex}] (0,0) -- (4.5,0) node[below] {$T$ (Força Aplicada)};
            \draw[-{Latex}] (0,0) -- (0,3.5) node[left] {$f$ (Atrito)};
            \draw[ultra thick,cyandark] (0,0) -- (2.5,2.5) node[midway,above,sloped] {\small Estático ($f_s=T$)};
            \draw[dashed] (2.5,0) -- (2.5,2.5) node[above] {\small $(f_s)_{\text{máx}}$};
            \draw[ultra thick,redaccent] (2.5,1.8) -- (4.2,1.8) node[right] {\small Cinético ($f_c = \mu_c n$)};
            \draw[dotted] (2.5,2.5) -- (2.5,1.8);
        \end{tikzpicture}
        \end{center}
    \end{columns}
\end{frame}

% --- SLIDE 12: EXEMPLO 5.13 & 5.14 - COEFICIENTES DE ATRITO ---
\begin{frame}{Exemplo 5.13 \& 5.14: Atrito Estático e Cinético em Engradado}
    \begin{block}{Enunciado do Problema}
        Um engradado de peso $p = 500\text{ N}$ exige uma força horizontal de $230\text{ N}$ para iniciar o movimento e $200\text{ N}$ para mantê-lo com velocidade constante. Ache $\mu_s$ e $\mu_c$. Qual a força de atrito se aplicarmos $50\text{ N}$ com ele em repouso?
    \end{block}

    \begin{columns}
        \column{0.45\textwidth}
        \begin{center}
        \begin{tikzpicture}[scale=0.8]
            \fill[navyblue] (2,1.5) circle (0.15);
            \draw[ultra thick,-{Latex},emeraldgreen] (2,1.5) -- (3.5,1.5) node[above] {\small $T = 230\text{ N}$};
            \draw[ultra thick,-{Latex},redaccent] (2,1.5) -- (0.5,1.5) node[above] {\small $(f_s)_{\text{máx}}$};
            \draw[ultra thick,-{Latex},cyandark] (2,1.5) -- (2,3.0) node[right] {\small $n = 500\text{ N}$};
            \draw[ultra thick,-{Latex},golddark] (2,1.5) -- (2,0.0) node[right] {\small $p = 500\text{ N}$};
            \node at (2,-0.4) {\small DCL no Limite};
        \end{tikzpicture}
        \end{center}

        \column{0.55\textwidth}
        \begin{exampleblock}{Resolução}
            \begin{enumerate}
                \item **Coeficiente Estático $\mu_s$**:
                $$\mu_s = \frac{(f_s)_{\text{máx}}}{n} = \frac{230\text{ N}}{500\text{ N}} = 0{,}46$$
                \item **Coeficiente Cinético $\mu_c$**:
                $$\mu_c = \frac{f_c}{n} = \frac{200\text{ N}}{500\text{ N}} = 0{,}40$$
                \item **Atrito com $T = 50\text{ N}$**:
                Como $50\text{ N} < 230\text{ N}$, repouso se mantém $\implies f_s = 50\text{ N}$.
            \end{enumerate}
        \end{exampleblock}
    \end{columns}
\end{frame}

% --- SLIDE 13: EXEMPLO 5.16 & 5.17 - TOBOGÃ COM ATRITO ---
\begin{frame}{Exemplo 5.16 \& 5.17: Tobogã em Plano Inclinado com Atrito}
    \begin{block}{Enunciado do Problema}
        Deduza o ângulo $\alpha$ para um tobogã deslizar com velocidade constante em uma encosta com coeficiente de atrito cinético $\mu_c$. Em uma encosta mais íngreme de ângulo $\alpha$, qual é a sua aceleração $a_x$?
    \end{block}

    \begin{columns}
        \column{0.5\textwidth}
        \begin{block}{Velocidade Constante ($\vec{a}=0$)}
            $$\sum F_x = mg \sin\alpha - f_c = 0$$
            $$\sum F_y = n - mg \cos\alpha = 0$$
            Como $f_c = \mu_c n = \mu_c mg \cos\alpha$:
            $$\tan\alpha = \mu_c \implies \alpha = \arctan\mu_c$$
        \end{block}

        \column{0.5\textwidth}
        \begin{block}{Movimento Acelerado ($\vec{a} \neq 0$)}
            $$\sum F_x = mg \sin\alpha - \mu_c mg \cos\alpha = m a_x$$
            Isolando a aceleração $a_x$:
            $$a_x = g (\sin\alpha - \mu_c \cos\alpha)$$
        \end{block}
    \end{columns}
\end{frame}

% --- SLIDE 14: EXEMPLO 5.19 - VELOCIDADE TERMINAL ---
\begin{frame}{Exemplo 5.19: Velocidade Terminal de um Pára-Quedista}
    \begin{block}{Enunciado do Problema}
        Para um corpo humano de $m = 50{,}0\text{ kg}$ caindo no ar em posição horizontal, o coeficiente de arraste turbulento é $D \approx 0{,}25\text{ kg/m}$. Ache a sua velocidade terminal $v_t$ em $\text{m/s}$ e em $\text{km/h}$.
    \end{block}

    \begin{columns}
        \column{0.45\textwidth}
        \begin{center}
        \begin{tikzpicture}[scale=0.85]
            \fill[navyblue] (1,2) circle (0.15);
            \draw[ultra thick,-{Latex},emeraldgreen] (1,2) -- (1,3.3) node[above] {$\vec{f}_{\text{arraste}} = D v_t^2$};
            \draw[ultra thick,-{Latex},redaccent] (1,2) -- (1,0.5) node[below] {$\vec{p} = m\vec{g}$};
            \node at (1,-0.3) {\small DCL na Velocidade Terminal};
        \end{tikzpicture}
        \end{center}

        \column{0.55\textwidth}
        \begin{exampleblock}{Resolução}
            Na velocidade terminal, a aceleração é nula ($a_y = 0$):
            $$\sum F_y = mg - D v_t^2 = 0 \implies D v_t^2 = mg$$
            $$v_t = \sqrt{\frac{mg}{D}} = \sqrt{\frac{(50{,}0\text{ kg})(9{,}80\text{ m/s}^2)}{0{,}25\text{ kg/m}}} = \sqrt{1960} = 44{,}3\text{ m/s}$$
            Convertendo para $\text{km/h}$:
            $$v_t = 44{,}3 \times 3{,}6 \approx 160\text{ km/h}$$
        \end{exampleblock}
    \end{columns}
\end{frame}

% --- SLIDE 15: 5.4 MOVIMENTO CIRCULAR UNIFORME ---
\begin{frame}{5.4 Dinâmica do Movimento Circular Uniforme}
    \begin{columns}
        \column{0.55\textwidth}
        \begin{block}{Aceleração Centrípeta / Radial}
            No MCU, a velocidade escalar $v$ é constante, mas a **direção do vetor velocidade muda continuamente**:

\begin{frame}{ALERTA DIDÁTICO: A Falácia da "Força Centrífuga"}
    \begin{columns}
        \column{0.5\textwidth}
        \begin{alertblock}{Por que NÃO existe força centrífuga?}
            \begin{itemize}
                \item Em um \textbf{referencial inercial}, não existe nenhuma força puxando o corpo para fora do círculo!
                \item A tendência do corpo de sair pela tangente é consequência da sua \textbf{inércia} ($1^{\text{a}}$ Lei de Newton).
                \item A quantidade $m \frac{v^2}{R}$ é o produto $m a_{\text{rad}}$, \textbf{NUNCA uma força aplicada} no DCL.
            \end{itemize}
        \end{alertblock}

% --- SLIDE 17: EXEMPLO 5.22 & 5.23 - CURVAS PLANAS E INCLINADAS ---
\begin{frame}{Exemplo 5.22 \& 5.23: Curva Plana vs Curva Inclinada}
    \begin{columns}
        \column{0.5\textwidth}
        \begin{block}{Exemplo 5.22: Curva Plana com Atrito ($\mu_s$)}
            \textbf{Enunciado}: Ache a velocidade máxima sem derrapar em curva plana de raio $R$ com atrito $\mu_s$.
            $$f_s = m \frac{v^2}{R} \le \mu_s mg \implies v_{\text{máx}} = \sqrt{\mu_s g R}$$
        \end{block}

        \column{0.5\textwidth}
        \begin{block}{Exemplo 5.23: Curva Inclinada sem Atrito ($\beta$)}
            \textbf{Enunciado}: Qual o ângulo de inclinação $\beta$ para fazer a curva com velocidade $v$ sem atrito?
            $$n \sin\beta = m \frac{v^2}{R} \quad \text{e} \quad n \cos\beta = mg$$
            $$\tan\beta = \frac{v^2}{gR}$$
        \end{block}
    \end{columns}
\end{frame}

% --- SLIDE 18: EXEMPLO 5.24 - RODA-GIGANTE EM PLANO VERTICAL ---
\begin{frame}{Exemplo 5.24: Movimento Circular em Círculo Vertical}
    \begin{block}{Enunciado do Problema}
        Um passageiro na roda-gigante de raio $R$ move-se em um círculo vertical com velocidade constante $v$. Deduza expressões para a força normal exercida pelo assento no topo do círculo e no ponto inferior (base).
    \end{block}

    \begin{columns}
        \column{0.5\textwidth}
        \begin{block}{Topo da Trajetória ($a_y = -v^2/R$)}
            $$n_T - mg = -m \frac{v^2}{R} \implies n_T = m\left(g - \frac{v^2}{R}\right)$$
            Sensação de peso menor no topo!
        \end{block}

        \column{0.5\textwidth}
        \begin{block}{Base da Trajetória ($a_y = +v^2/R$)}
            $$n_B - mg = +m \frac{v^2}{R} \implies n_B = m\left(g + \frac{v^2}{R}\right)$$
            Sensação de peso maior na base ($n_B > mg$)!
        \end{block}
    \end{columns}
\end{frame}
